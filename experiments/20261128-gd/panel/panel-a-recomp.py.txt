def load(s,c):
    z=np.load(B+f'{s}/{c}/errors.npz'); return z
def eng_logs(z,ctl,sysn,col=0,rows=None):
    o=z['ordinal']; a=z['err_'+ctl][:,col]; b=z['err_'+sysn][:,col]
    ok=np.isfinite(a)&np.isfinite(b)
    if rows is not None: ok&=rows
    d={}
    for u in np.unique(o[ok]):
        m=ok&(o==u); x=np.median(a[m]); y=np.median(b[m])
        if x>0 and y>0: d[int(u)]=math.log(x/y)
    return d
def pooled(z,ctl,systems,col=0,rows=None):
    per=[eng_logs(z,ctl,s,col,rows) for s in systems]
    com=sorted(set.intersection(*[set(p) for p in per]))
    return {u:np.mean([p[u] for p in per]) for u in com}
def boot_ratio(d,seed=1,R=4000):
    x=np.array(list(d.values())); r=np.random.default_rng(seed)
    m=x[r.integers(0,x.size,(R,x.size))].mean(1)
    return math.exp(x.mean()),math.exp(np.quantile(m,.025)),math.exp(np.quantile(m,.975)),x.size
def boot_share(n,d,seed=2,R=4000):
    com=sorted(set(n)&set(d)); a=np.array([n[u] for u in com]); b=np.array([d[u] for u in com])
    r=np.random.default_rng(seed); p=r.integers(0,a.size,(R,a.size))
    s=a[p].mean(1)/b[p].mean(1)
    return a.mean()/b.mean(),np.quantile(s,.025),np.quantile(s,.975),a.size
