rng=np.random.default_rng(0)
def eng(d,sys,q,mask):
    o=d['ordinal']; out={}
    for u in np.unique(o):
        m=(o==u)&mask
        if m.sum(): out[u]=np.median(np.abs(d['err_'+sys][m,q]))
    return out
def ratio(d,a,b,q=0,mask=None,ft=False):
    n=len(d['ordinal']); mask=np.ones(n,bool) if mask is None else mask
    bs=[b] if not ft else ['ft-s1','ft-s2','ft-s3']
    lr=[]
    fin=mask&np.isfinite(d['err_'+a][:,q])
    for s in bs: fin&=np.isfinite(d['err_'+s][:,q])
    ea=eng(d,a,q,fin); 
    L=np.mean([[np.log(ea[u]/eng(d,s,q,fin)[u]) for u in ea] for s in bs],axis=0)
    L=L[np.isfinite(L)]
    bt=[np.mean(rng.choice(L,len(L))) for _ in range(500)]
    return "%.3f [%.3f,%.3f] med %.2f n%d b%.2f 2g%.2f 2b%.2f"%(np.exp(L.mean()),*np.exp(np.percentile(bt,[2.5,97.5])),np.exp(np.median(L)),len(L),(L>0).mean(),(L>np.log(2)).mean(),(L<-np.log(2)).mean())
