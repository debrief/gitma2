def englog(d,ctl,sys,col=0,rows=None,cap=None,stat=np.median):
    a=d['err_'+ctl][:,col].copy(); b=d['err_'+sys][:,col].copy()
    if cap is not None:
        a[~np.isfinite(a)]=cap; b[~np.isfinite(b)]=cap
    ok=np.isfinite(a)&np.isfinite(b)
    if rows is not None: ok&=rows
    o=d['ordinal']; out={}
    for u in np.unique(o[ok]):
        m=ok&(o==u); x,y=stat(a[m]),stat(b[m])
        if x>0 and y>0: out[int(u)]=math.log(x/y)
    return out
def pooled(d,ctl,syss,**kw):

    com=sorted(set.intersection(*(set(p) for p in per)))
    return {u:float(np.mean([p[u] for p in per])) for u in com}
def boot(x,f=np.mean,n=2000,seed=1,level=.95):
    r=np.random.default_rng(seed); idx=r.integers(0,len(x),(n,len(x)))
    v=np.array([f(x[i]) for i in idx]) if f is not np.mean else x[idx].mean(1)
    return np.quantile(v,[(1-level)/2,(1+level)/2])
def tmean(x,p):
    x=np.sort(x); k=int(len(x)*p); return x[k:len(x)-k].mean()
def share(num,den,f=np.mean,seed=41,n=2000,level=.95):

    r=np.random.default_rng(seed); idx=r.integers(0,len(a),(n,len(a)))
    if f is np.mean: v=a[idx].mean(1)/b[idx].mean(1)
    else: v=np.array([f(a[i])/f(b[i]) for i in idx])
    lo,hi=np.quantile(v,[(1-level)/2,(1+level)/2]); return f(a)/f(b),lo,hi,len(a)
