import numpy as np
def gmed(x,o):
    i=np.argsort(o,kind='stable'); o=o[i]; x=x[i]; u,st=np.unique(o,return_index=True)
    return np.array([np.median(c) for c in np.split(x,st[1:])])

rng=np.random.default_rng(0)
FT=['ft-s1','ft-s2','ft-s3']
def L(d,a,bs,q=0,mask=None):
    m=np.ones(len(d['k']),bool) if mask is None else mask.copy()
    for s in [a]+bs: m&=np.isfinite(d['err_'+s][:,q])
    g={s:gmed(np.abs(d['err_'+s][m,q]),d['ordinal'][m]) for s in [a]+bs}
    l=np.mean([np.log(g[a]/g[s]) for s in bs],axis=0); return l[np.isfinite(l)]
def R(d,a,bs,q=0,mask=None):
    l=L(d,a,bs,q,mask); bt=np.exp([rng.choice(l,len(l)).mean() for _ in range(1000)])
