import numpy as np, sys
S=sys.argv[1]
def lr(z,num,den,c):
    a=z['err_'+num][:,c]; b=z['err_'+den][:,c]; o=z['ordinal']
    ok=np.isfinite(a)&np.isfinite(b); d={}
    for u in np.unique(o[ok]):
        m=ok&(o==u); x=np.median(a[m]); y=np.median(b[m])
        if x>0 and y>0: d[int(u)]=np.log(x/y)
    return d
def pooled(z,pairs,c,drop_first=False):
    ds=[lr(z,n,d,c) for n,d in pairs]
    com=sorted(set.intersection(*[set(d) for d in ds]))
    v=np.array([np.mean([d[u] for d in ds]) for u in com])
    rng=np.random.default_rng(12345)
    bs=v[rng.integers(0,len(v),(4000,len(v)))].mean(1)
    return len(v),np.exp(v.mean()),np.exp(np.quantile(bs,[.025,.975])),np.exp(np.median(v)),(v>0).mean(),(v<-np.log(2)).mean()
