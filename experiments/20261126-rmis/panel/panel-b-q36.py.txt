def boot(lr,seed,B):
    n=lr.size; rng=np.random.default_rng(seed)
    m=np.concatenate([lr[rng.integers(0,n,(2000,n))].mean(1) for _ in range(B//2000)])
    return np.exp(np.quantile(m,[.025,.975]))
